# TensorRT-LLM ASR runtime dependencies for the (Indic) Canary stage
# (nemo_curator.stages.audio.inference.indic_canary).
#
# These are intentionally NOT part of the `audio_cuda12` extra / uv.lock:
# tensorrt_llm pins (transformers, wheel, pillow, torch) conflict with Curator's
# stack (nemo_toolkit needs transformers>=4.57), so co-resolving them is
# impossible. This runtime is installed as a separate step on top of the venv.
#
# Install (into the Curator venv):
#   uv pip install -r requirements-trt-llm.txt
# or:
#   pip install -r requirements-trt-llm.txt
#
# Linux x86_64 only -- no macOS / aarch64 wheels are published.
# Pinned to 1.2.1 (CUDA 13 line, cp312 wheels) to match a CUDA-13 base image
# such as pytorch/pytorch:2.9.0-cuda13.0. NOTE: engines built with the 0.20.x
# (TensorRT 10.9) toolchain must be rebuilt to match this runtime's TensorRT.

--extra-index-url https://pypi.nvidia.com
datasets
kaldialign
lhotse
librosa
onnx_graphsurgeon
onnxscript
pytorch-lightning>=2.4.0
safetensors

# --- Inference runtime (required by the Curator stage) ---
tensorrt_llm==1.2.1
tensorrt_llm>=0.17.0

# --- Offline engine-build toolchain (only needed to convert a .nemo -> TRT-LLM
#     engine via convert_checkpoint.py / conformer_onnx_trt.py / trtllm-build;
#     not needed to run inference against a prebuilt engine) ---
torch>=2.5.0
torchtune
